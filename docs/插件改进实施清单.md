#### 3.2 数据分析模块
**新建文件**：`astrbot/plugins/common/analytics.py`

```python
from datetime import datetime, timedelta
from typing import List, Dict

class PluginAnalytics:
    def __init__(self, db_manager):
        self.db = db_manager
    
    async def log_search(self, user_id: str, keyword: str, 
                        resource_type: str, found: bool, 
                        result_count: int = 0):
        """记录搜索行为"""
        await self.db.execute(
            """
            INSERT INTO search_logs 
            (user_id, keyword, resource_type, found, result_count, timestamp)
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (user_id, keyword, resource_type, found, result_count, datetime.now())
        )
    
    async def get_hot_keywords(self, resource_type: str = None, 
                              days: int = 7, limit: int = 10) -> List[Dict]:
        """获取热门搜索词"""
        query = """
            SELECT keyword, COUNT(*) as count
            FROM search_logs
            WHERE timestamp >= ?
        """
        params = [datetime.now() - timedelta(days=days)]
        
        if resource_type:
            query += " AND resource_type = ?"
            params.append(resource_type)
        
        query += " GROUP BY keyword ORDER BY count DESC LIMIT ?"
        params.append(limit)
        
        results = await self.db.execute(query, params)
        return [{"keyword": r[0], "count": r[1]} for r in results]
    
    async def get_not_found_keywords(self, days: int = 7, 
                                    limit: int = 20) -> List[Dict]:
        """获取未命中的搜索词（用于指导资源补充）"""
        query = """
            SELECT keyword, resource_type, COUNT(*) as count
            FROM search_logs
            WHERE found = 0 AND timestamp >= ?
            GROUP BY keyword, resource_type
            ORDER BY count DESC
            LIMIT ?
        """
        results = await self.db.execute(
            query,
            (datetime.now() - timedelta(days=days), limit)
        )
        return [
            {"keyword": r[0], "resource_type": r[1], "count": r[2]} 
            for r in results
        ]
```
