# 知识猎人 - 通用配额系统设计方案

## 🎯 系统目标

设计一套**积分+会员+配额**三位一体的通用系统，支持：
- ✅ 多插件统一管理
- ✅ 灵活的配额规则配置
- ✅ 积分充值和消费
- ✅ 会员等级权益
- ✅ 动态配额调整
- ✅ 易于扩展新插件

## 🗄️ 数据库设计

### 1. 用户表 (users)
```sql
CREATE TABLE users (
    user_id TEXT PRIMARY KEY,              -- 用户ID
    username TEXT,                         -- 用户名
    platform TEXT NOT NULL,                -- 平台 (telegram/lark/qq等)
    platform_user_id TEXT NOT NULL,        -- 平台用户ID
    created_at DATETIME NOT NULL,          -- 注册时间
    last_active_at DATETIME,               -- 最后活跃时间
    is_banned BOOLEAN DEFAULT 0,           -- 是否封禁
    ban_reason TEXT,                       -- 封禁原因
    INDEX idx_platform_user (platform, platform_user_id)
);
```

---

### 2. 会员表 (memberships)
```sql
CREATE TABLE memberships (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id TEXT NOT NULL,                 -- 用户ID
    level INTEGER NOT NULL DEFAULT 0,      -- 会员等级 (0=免费, 1=高级, 2=VIP)
    expire_date DATE,                      -- 到期日期 (NULL表示永久或免费)
    auto_renew BOOLEAN DEFAULT 0,          -- 是否自动续费
    created_at DATETIME NOT NULL,          -- 开通时间
    updated_at DATETIME NOT NULL,          -- 更新时间
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    INDEX idx_user_id (user_id),
    INDEX idx_expire_date (expire_date)
);
```

### 3. 积分账户表 (points_accounts)
```sql
CREATE TABLE points_accounts (

    total_earned INTEGER DEFAULT 0,        -- 累计获得积分
    total_spent INTEGER DEFAULT 0,         -- 累计消费积分
    created_at DATETIME NOT NULL,          -- 创建时间
    updated_at DATETIME NOT NULL,          -- 更新时间
    FOREIGN KEY (user_id) REFERENCES users(user_id)
```

### 4. 积分流水表 (points_transactions)
```sql
CREATE TABLE points_transactions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,

    amount INTEGER NOT NULL,               -- 积分变动量 (正数=获得, 负数=消费)

    FOREIGN KEY (user_id) REFERENCES users(user_id),
    INDEX idx_user_id (user_id),
    INDEX idx_type (type),
    INDEX idx_created_at (created_at)
);
```

### 5. 配额规则表 (quota_rules)
```sql
CREATE TABLE quota_rules (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    action_type TEXT NOT NULL,             -- 操作类型 (如 music_download_flac)
    plugin_name TEXT NOT NULL,             -- 插件名称 (如 music)

    is_active BOOLEAN DEFAULT 1,           -- 是否启用
    created_at DATETIME NOT NULL,          -- 创建时间
    updated_at DATETIME NOT NULL,          -- 更新时间
    UNIQUE(action_type, member_level),
    INDEX idx_plugin (plugin_name),
    INDEX idx_action (action_type)
);
```

### 1. 配额验证器 (QuotaValidator)

```python
# data/plugins/common/quota_validator.py

from enum import IntEnum
from typing import Optional, Tuple
from datetime import date, datetime
import sqlite3

class MemberLevel(IntEnum):
    """会员等级"""
    FREE = 0      # 免费用户
    PREMIUM = 1   # 高级会员 ¥19.9/月
    VIP = 2       # VIP会员 ¥399/年

class QuotaResult:
    """配额检查结果"""
    def __init__(self, allowed: bool, message: str = "", 
                 remaining: int = 0, points_cost: int = 0):
        self.allowed = allowed          # 是否允许
        self.message = message          # 提示消息
        self.remaining = remaining      # 剩余次数
        self.points_cost = points_cost  # 积分消耗

class QuotaValidator:
    """配额验证器 - 统一管理所有插件的配额"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self._init_database()
    
    def _init_database(self):
        """初始化数据库表"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 创建所有表（省略具体SQL，见上文数据库设计）
        # ...
        
        conn.commit()
        conn.close()
    
    async def check_quota(
        self, 
        user_id: str, 
        action_type: str,
        plugin_name: str,
        use_points: bool = False
    ) -> QuotaResult:
        """
        检查用户配额
        
        Args:
            user_id: 用户ID
            action_type: 操作类型 (如 music_download_flac)
            plugin_name: 插件名称 (如 music)
            use_points: 是否使用积分抵扣
            
        Returns:
            QuotaResult 配额检查结果
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        try:
            # 1. 获取用户会员等级
            member_level = self._get_member_level(cursor, user_id)
            
            # 2. 获取配额规则
            rule = self._get_quota_rule(cursor, action_type, member_level)
            if not rule:
                # 没有配置规则，默认允许
                return QuotaResult(allowed=True, message="操作允许")
            
            daily_limit, points_cost = rule
            
            # 3. 检查是否无限制
            if daily_limit == -1:
                return QuotaResult(
                    allowed=True, 
                    message="VIP用户无限制",
                    remaining=-1
                )
            
            # 4. 查询今日已使用次数
            today = date.today()
            used_count = self._get_today_usage(cursor, user_id, action_type, today)
            
            # 5. 查询临时加成
            boost_amount = self._get_active_boosts(cursor, user_id, action_type, today)
            
            # 6. 计算可用配额
            total_limit = daily_limit + boost_amount
            remaining = total_limit - used_count
            
            # 7. 判断是否超限
            if remaining <= 0:
                # 超限，检查是否可以用积分抵扣
                if use_points and points_cost > 0:
                    points_balance = self._get_points_balance(cursor, user_id)
                    if points_balance >= points_cost:
                        # 积分足够，允许操作
                        return QuotaResult(
                            allowed=True,
                            message=f"使用{points_cost}积分抵扣",
                            remaining=0,
                            points_cost=points_cost
                        )
                    else:
                        return QuotaResult(
                            allowed=False,
                            message=f"今日配额已用完，积分不足（需要{points_cost}，当前{points_balance}）",
                            remaining=0
                        )
                else:
                    # 生成升级提示
                    upgrade_msg = self._generate_upgrade_message(
                        action_type, member_level, daily_limit
                    )
                    return QuotaResult(
                        allowed=False,
                        message=upgrade_msg,
                        remaining=0
                    )
            
            # 8. 配额充足，允许操作
            return QuotaResult(
                allowed=True,
                message=f"操作允许，今日剩余{remaining}次",
                remaining=remaining
            )
            
        finally:
            conn.close()
    
    async def consume_quota(
        self, 
        user_id: str, 
        action_type: str,
        plugin_name: str,
        points_cost: int = 0
    ) -> bool:
        """
        消费配额
        
        Args:
            user_id: 用户ID
            action_type: 操作类型
            plugin_name: 插件名称
            points_cost: 积分消耗
            
        Returns:
            是否成功
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        try:
            today = date.today()
            
            # 1. 记录配额使用
            cursor.execute("""
                INSERT INTO quota_usage 
                (user_id, action_type, plugin_name, usage_date, count, points_spent, created_at)
                VALUES (?, ?, ?, ?, 1, ?, ?)
            """, (user_id, action_type, plugin_name, today, points_cost, datetime.now()))
            
            # 2. 如果消耗积分，扣除积分
            if points_cost > 0:
                self._deduct_points(cursor, user_id, points_cost, action_type)
            
            conn.commit()
            return True
            
        except Exception as e:
            conn.rollback()
            print(f"消费配额失败: {e}")
            return False
        finally:
            conn.close()
    
    def _get_member_level(self, cursor, user_id: str) -> MemberLevel:
        """获取用户会员等级"""
        cursor.execute("""
            SELECT level, expire_date 
            FROM memberships 
            WHERE user_id = ?
        """, (user_id,))
        
        row = cursor.fetchone()
        if not row:
            return MemberLevel.FREE
        
        level, expire_date = row
        
        # 检查是否过期
        if expire_date:
            expire = datetime.strptime(expire_date, "%Y-%m-%d").date()
            if expire < date.today():
                return MemberLevel.FREE
        
        return MemberLevel(level)
    
    def _get_quota_rule(self, cursor, action_type: str, member_level: MemberLevel) -> Optional[Tuple[int, int]]:
        """获取配额规则"""
        cursor.execute("""
            SELECT daily_limit, points_cost
            FROM quota_rules
            WHERE action_type = ? AND member_level = ? AND is_active = 1
        """, (action_type, member_level.value))
        
        row = cursor.fetchone()
        return row if row else None
    
    def _get_today_usage(self, cursor, user_id: str, action_type: str, today: date) -> int:
        """获取今日已使用次数"""
        cursor.execute("""
            SELECT COALESCE(SUM(count), 0)
            FROM quota_usage
            WHERE user_id = ? AND action_type = ? AND usage_date = ?
        """, (user_id, action_type, today))
        
        return cursor.fetchone()[0]
    
    def _get_active_boosts(self, cursor, user_id: str, action_type: str, today: date) -> int:
        """获取有效的临时加成"""
        cursor.execute("""
            SELECT COALESCE(SUM(boost_amount), 0)
            FROM quota_boosts
            WHERE user_id = ? 
              AND (action_type = ? OR action_type IS NULL)
              AND expire_date >= ?
              AND is_used = 0
        """, (user_id, action_type, today))
        
        return cursor.fetchone()[0]
    
    def _get_points_balance(self, cursor, user_id: str) -> int:
        """获取积分余额"""
        cursor.execute("""
            SELECT balance FROM points_accounts WHERE user_id = ?
        """, (user_id,))
        
        row = cursor.fetchone()
        return row[0] if row else 0
    
    def _deduct_points(self, cursor, user_id: str, amount: int, description: str):
        """扣除积分"""
        # 1. 更新余额
        cursor.execute("""
            UPDATE points_accounts 
            SET balance = balance - ?,
                total_spent = total_spent + ?,
                updated_at = ?
            WHERE user_id = ?
        """, (amount, amount, datetime.now(), user_id))
        
        # 2. 记录流水
        cursor.execute("""
            SELECT balance FROM points_accounts WHERE user_id = ?
        """, (user_id,))
        balance_after = cursor.fetchone()[0]
        
        cursor.execute("""
            INSERT INTO points_transactions
            (user_id, amount, balance_after, type, source, description, created_at)
            VALUES (?, ?, ?, 'consume', 'quota', ?, ?)
        """, (user_id, -amount, balance_after, description, datetime.now()))
    
    def _generate_upgrade_message(self, action_type: str, member_level: MemberLevel, daily_limit: int) -> str:
        """生成升级提示消息"""
        action_name_map = {
            "music_download_flac": "无损音乐下载",
            "music_download_320": "320k音乐下载",
            "yunpan_download": "云盘资源下载",
            "douban_view": "豆瓣评分查看"
        }
        
        action_name = action_name_map.get(action_type, "该操作")
        
        if member_level == MemberLevel.FREE:
            return f"""
⚠️ 今日{action_name}次数已用完
免费用户每日限{daily_limit}次

💎 升级会员享受：
• 高级会员(¥19.9/月): 每日{daily_limit * 10}次
• VIP会员(¥399/年): 无限次数
• 无广告体验
• 优先获取资源
• 专属客服支持

💰 或使用积分抵扣：

• 充值积分: /充值

👉 立即升级: /upgrade
"""
        else:
            return f"""

高级会员每日限{daily_limit}次
```

### 2. 积分管理器 (PointsManager)

```python
# data/plugins/common/points_manager.py

from datetime import datetime
import sqlite3

class PointsManager:
    """积分管理器"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    async def recharge(
        self, 
        user_id: str, 
        amount: int, 
        source: str = "payment",
        order_id: str = None,
        description: str = "积分充值"
    ) -> bool:
        """
        充值积分
        
        Args:
            user_id: 用户ID
            amount: 充值金额
            source: 来源 (payment/admin/gift)
            order_id: 订单ID
            description: 描述
            
        Returns:
            是否成功
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        try:
            # 1. 确保用户有积分账户
            cursor.execute("""
                INSERT OR IGNORE INTO points_accounts 
                (user_id, balance, total_earned, total_spent, created_at, updated_at)
                VALUES (?, 0, 0, 0, ?, ?)
            """, (user_id, datetime.now(), datetime.now()))
            
            # 2. 更新余额
            cursor.execute("""
                UPDATE points_accounts 
                SET balance = balance + ?,
                    total_earned = total_earned + ?,
                    updated_at = ?
                WHERE user_id = ?
            """, (amount, amount, datetime.now(), user_id))
            
            # 3. 获取更新后余额
            cursor.execute("""
                SELECT balance FROM points_accounts WHERE user_id = ?
            """, (user_id,))
            balance_after = cursor.fetchone()[0]
            
            # 4. 记录流水
            cursor.execute("""
                INSERT INTO points_transactions
                (user_id, amount, balance_after, type, source, description, related_order_id, created_at)
                VALUES (?, ?, ?, 'recharge', ?, ?, ?, ?)
            """, (user_id, amount, balance_after, source, description, order_id, datetime.now()))
            
            conn.commit()
            return True
            
        except Exception as e:
            conn.rollback()
            print(f"充值积分失败: {e}")
            return False
        finally:
            conn.close()
    
    async def get_balance(self, user_id: str) -> int:
        """获取积分余额"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT balance FROM points_accounts WHERE user_id = ?
        """, (user_id,))
        
        row = cursor.fetchone()
        conn.close()
        
        return row[0] if row else 0
    
    async def add_boost(
        self,
        user_id: str,
        action_type: str,
        boost_amount: int,
        days: int = 1,
        source: str = "points",
        description: str = "积分兑换配额加成"
    ) -> bool:
        """
        添加临时配额加成
        
        Args:
            user_id: 用户ID
            action_type: 操作类型 (None表示全局)
            boost_amount: 加成数量
            days: 有效天数
            source: 来源
            description: 描述
            
        Returns:
            是否成功
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        try:
            from datetime import timedelta
            expire_date = (datetime.now() + timedelta(days=days)).date()
            
            cursor.execute("""
                INSERT INTO quota_boosts
                (user_id, action_type, boost_amount, expire_date, source, description, created_at)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (user_id, action_type, boost_amount, expire_date, source, description, datetime.now()))
            
            conn.commit()
            return True
            
        except Exception as e:
            conn.rollback()
            print(f"添加配额加成失败: {e}")
            return False
        finally:
            conn.close()
    
    async def get_transactions(self, user_id: str, limit: int = 10):
        """获取积分流水"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT amount, balance_after, type, source, description, created_at
            FROM points_transactions
            WHERE user_id = ?
            ORDER BY created_at DESC
            LIMIT ?
        """, (user_id, limit))
        
        rows = cursor.fetchall()
        conn.close()
        
        return rows
```

### 3. 会员管理器 (MembershipManager)

```python
# data/plugins/common/membership_manager.py

from datetime import datetime, date, timedelta
import sqlite3
from .quota_validator import MemberLevel

class MembershipManager:
    """会员管理器"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    async def upgrade(
        self,
        user_id: str,
        level: MemberLevel,
        months: int = 1
    ) -> bool:
        """
        升级会员
        
        Args:
            user_id: 用户ID
            level: 会员等级
            months: 月数
            
        Returns:
            是否成功
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        try:
            # 1. 查询现有会员信息
            cursor.execute("""
                SELECT level, expire_date FROM memberships WHERE user_id = ?
            """, (user_id,))
            
            row = cursor.fetchone()
            
            if row:
                current_level, current_expire = row
                current_expire = datetime.strptime(current_expire, "%Y-%m-%d").date() if current_expire else date.today()
                
                # 如果当前会员未过期，从到期日开始续期
                if current_expire > date.today():
                    new_expire = current_expire + timedelta(days=30 * months)
                else:
                    new_expire = date.today() + timedelta(days=30 * months)
                
                # 更新会员信息
                cursor.execute("""
                    UPDATE memberships
                    SET level = ?, expire_date = ?, updated_at = ?
                    WHERE user_id = ?
                """, (level.value, new_expire, datetime.now(), user_id))
            else:
                # 创建新会员
                new_expire = date.today() + timedelta(days=30 * months)
                cursor.execute("""
                    INSERT INTO memberships
                    (user_id, level, expire_date, created_at, updated_at)
                    VALUES (?, ?, ?, ?, ?)
                """, (user_id, level.value, new_expire, datetime.now(), datetime.now()))
            
            conn.commit()
            return True
            
        except Exception as e:
            conn.rollback()
            print(f"升级会员失败: {e}")
            return False
        finally:
            conn.close()
```

```python
# astrbot_plugin_music/handlers/download_handler.py

async def handle_download(self, user_id, song_type="flac"):
    # 1. 检查配额
    result = await self.quota_validator.check_quota(
        user_id, 
        action_type=f"music_download_{song_type}",
        plugin_name="music",
        use_points=True  # 允许使用积分
    )

        # 配额不足且积分不足，返回提示
        return result.message

    success = await self.download_service.download(song_id)
    
    if success:
        # 3. 扣减配额（和积分）
        await self.quota_validator.consume_quota(
            user_id, 
            action_type=f"music_download_{song_type}",
            plugin_name="music",
            points_cost=result.points_cost
        )
```
