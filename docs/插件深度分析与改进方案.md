# 知识猎人插件深度分析与改进方案

## 📊 现有插件功能分析

### 1. astrbot_plugin_douban（豆瓣评分插件）
**核心功能**：
- 自动识别豆瓣链接（电影/图书/电视剧）
- 生成豆瓣评分图片
- 获取热门用户评论（前4条）
- 提供"搜索资源"和"查看详情"按钮

**技术特点**：
- 支持桌面版、移动版、App调度链接
- 异步并行获取图片和评论数据
- 完善的URL解析正则表达式
- 调用第三方API：`https://api.wowoziyuan.com/douban/`

**存在问题**：
- ❌ 依赖第三方API，稳定性不可控
- ❌ 缺少会员权益差异化（所有用户功能一致）
- ❌ 无配额限制机制
- ❌ 缺少数据统计和用户行为追踪
- ❌ 评论内容未做版权声明

### 2. astrbot_plugin_music（音乐搜索插件）
**核心功能**：
- 通过 `/歌` 命令搜索音乐
- 支持多平台（QQ音乐、网易云等）
- 配额管理系统（128k/320k/flac等不同品质）

- VIP倍率机制（VIP用户配额x3）

**技术特点**：
- 完善的数据库设计（SQLite）
- 配额管理器（QuotaManager）
- 文件缓存管理器（FileCacheManager）
- 支持QQ音乐短链解析
- 歌词格式化处理（移除时间标签）
- Telegram Bot API集成

**存在问题**：
- ⚠️ 配额系统已实现，但需与会员体系深度绑定
- ⚠️ 缺少搜索未命中统计（用于指导资源补充）
- ⚠️ 无用户行为分析（搜索热词、下载热度）
- ⚠️ 缺少版权声明和正版导流
- ⚠️ 错误提示不够用户友好

```python
# astrbot/plugins/common/analytics.py

class PluginAnalytics:
    """插件数据分析"""
    
    def __init__(self, db_manager):
        self.db = db_manager
    
    async def log_search(self, user_id: str, keyword: str, 
                        resource_type: str, found: bool):
        """记录搜索行为"""
        await self.db.execute(
            """
            INSERT INTO search_logs 
            (user_id, keyword, resource_type, found, timestamp)
            VALUES (?, ?, ?, ?, ?)
            """,
            (user_id, keyword, resource_type, found, datetime.now())
        )
    
    async def log_download(self, user_id: str, resource_id: str, 
                          resource_type: str):
        """记录下载行为"""
        await self.db.execute(
            """
            INSERT INTO download_logs 
            (user_id, resource_id, resource_type, timestamp)
            VALUES (?, ?, ?, ?)
            """,
            (user_id, resource_id, resource_type, datetime.now())
        )
    
    async def get_hot_keywords(self, resource_type: str, 
                              days: int = 7, limit: int = 10) -> list:
        """获取热门搜索词"""
        result = await self.db.execute(
            """
            SELECT keyword, COUNT(*) as count
            FROM search_logs
            WHERE resource_type = ? 
            AND timestamp >= datetime('now', '-{} days')
            GROUP BY keyword
            ORDER BY count DESC
            LIMIT ?
            """.format(days),
            (resource_type, limit)
        )
        return result
    
    async def get_not_found_keywords(self, resource_type: str, 
                                    days: int = 7, limit: int = 20) -> list:
        """获取未命中的搜索词（用于指导资源补充）"""
        result = await self.db.execute(
            """
            SELECT keyword, COUNT(*) as count
            FROM search_logs
            WHERE resource_type = ? 
            AND found = 0
            AND timestamp >= datetime('now', '-{} days')
            GROUP BY keyword
            ORDER BY count DESC
            LIMIT ?
            """.format(days),
            (resource_type, limit)
        )
        return result
    
    async def get_user_behavior_profile(self, user_id: str) -> dict:
        """获取用户行为画像"""
        # 搜索偏好
        search_prefs = await self.db.execute(
            """
            SELECT resource_type, COUNT(*) as count
            FROM search_logs
            WHERE user_id = ?
            GROUP BY resource_type
            """,
            (user_id,)
        )
        
        # 下载偏好
        download_prefs = await self.db.execute(
            """
            SELECT resource_type, COUNT(*) as count
            FROM download_logs
            WHERE user_id = ?
            GROUP BY resource_type
            """,
            (user_id,)
        )
        
        return {
            "search_preferences": search_prefs,
            "download_preferences": download_prefs
        }
```

```python
class ResourceRecommender:
    """资源推荐系统"""
    
    def __init__(self, analytics: PluginAnalytics):
        self.analytics = analytics
    
    async def recommend_for_user(self, user_id: str, 
                                 resource_type: str, 
                                 limit: int = 5) -> list:
        """为用户推荐资源"""
        # 1. 获取用户行为画像

        # 2. 基于协同过滤推荐
        similar_users = await self._find_similar_users(user_id, profile)
        
        # 3. 获取相似用户喜欢的资源
        recommendations = await self._get_popular_resources_from_users(
            similar_users, resource_type, limit
        )
        
        return recommendations
    
    async def recommend_related_resources(self, resource_id: str, 
                                         resource_type: str) -> list:
        """推荐相关资源（跨插件联动）"""
        # 例如：搜索书籍时推荐相关音乐
        if resource_type == "book":
            # 提取书籍关键词（作者、主题）
            keywords = await self._extract_keywords(resource_id)
            # 搜索相关音乐
            related_music = await self._search_related_music(keywords)

        elif resource_type == "music":
            # 搜索音乐时推荐相关书籍
            keywords = await self._extract_keywords(resource_id)
            related_books = await self._search_related_books(keywords)
            return related_books
        
        return []
```

```python
from abc import ABC, abstractmethod

class BaseResourcePlugin(ABC):
    """资源插件基类"""
    
    def __init__(self, context: Context):
        self.context = context
        self.config = self._load_config()
        self.db = self._init_database()
        self.member_validator = MemberValidator()
        self.analytics = PluginAnalytics(self.db)
        self.copyright_mgr = CopyrightManager()
        self.formatter = MessageFormatter()
    
    @abstractmethod
    async def search(self, keyword: str, user_id: str) -> list:
        """搜索资源"""
        pass
    
    @abstractmethod
    async def get_detail(self, resource_id: str, user_id: str) -> dict:
        """获取资源详情"""
        pass
    
    @abstractmethod
    async def download(self, resource_id: str, user_id: str) -> str:
        """下载资源"""
        pass
    
    async def check_quota(self, user_id: str, action: str) -> bool:
        """检查配额（统一实现）"""
```
